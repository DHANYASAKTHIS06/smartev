Live Cypher Execution: 
MATCH (ev:EV {evId: 'EV-TN-01'})-[:LOCATED_AT]->(start:Location {name: 'Coimbatore'})
MATCH (dest:Location {name: 'Salem'})
MATCH path = shortestPath((start)-[:CONNECTED_TO*..5]->(dest))
RETURN path, reduce(dist = 0, r in relationships(path) | dist + r.distance) AS totalDistance,
       reduce(energy = 0, r in relationships(path) | energy + r.energyCost) AS totalEnergyRequired
